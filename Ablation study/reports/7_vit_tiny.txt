model_defs.py
-------------
SOTA Benchmark Encoder Module (Pretrained = True).
Master script: Automatically handles timm quirks for CNNs, Transformers, and Hybrids.
"""

import torch
import torch.nn as nn
import torch.nn.functional as F

try:
    import timm
except ImportError:
    raise ImportError("Please install timm using 'pip install timm' to use SOTA models.")


def get_patch_encoder(input_shape=(128, 128, 1), embedding_dim=128) -> nn.Module:
    """
    Builds the base patch encoder using modern SOTA architectures via timm.
    Outputs L2-normalized embeddings required for Triplet Loss.
    """
    class TimmPatchEncoder(nn.Module):
        def __init__(self):
            super().__init__()
            
            # =================================================================
            # --- SWAP YOUR MODELS HERE ---
            # 'resnet18', 'mobilenetv4_conv_small', 'ghostnetv2_100', 
            # 'fasternet_t0', 'edgenext_xx_small', 'efficientvit_m0', 
            # 'vit_tiny_patch16_224', 'convnextv2_atto'
            # =================================================================
            model_name = 'vit_tiny_patch16_224' 
            
            self.needs_interpolation = False

            # 1. EdgeNeXt handles spatial dims dynamically and rejects img_size
            if 'edgenext' in model_name:
                self.backbone = timm.create_model(
                    model_name, pretrained=True, in_chans=1, num_classes=embedding_dim
                )
            
            # 2. EfficientViT has rigid 49-param pretrained attention biases. 
            # It must be built at native 224x224, and we interpolate inputs to match.
            elif 'efficientvit' in model_name:
                self.backbone = timm.create_model(
                    model_name, pretrained=True, in_chans=1, num_classes=embedding_dim
                )
                self.needs_interpolation = True
            
            # 3. ViTs and Standard CNNs accept img_size perfectly.
            else:
                self.backbone = timm.create_model(
                    model_name, pretrained=True, in_chans=1, num_classes=embedding_dim, img_size=input_shape[0]
                )

        def forward(self, x):
            # Apply the interpolation hack ONLY if the specific model requires it
            if self.needs_interpolation and x.shape[-1] != 224:
                x = F.interpolate(x, size=(224, 224), mode='bilinear', align_corners=False)
                
            # Pass input through the backbone
            x = self.backbone(x)
            
            # Crucial for Triplet Loss: L2 Normalize the final embedding
            return F.normalize(x, p=2, dim=1)

    return TimmPatchEncoder()


def get_line_encoder(patch_encoder: nn.Module, patches_per_line: int=None) -> nn.Module:
    """
    Wraps the patch encoder to process an entire line (K patches) at once.
    """
    class FastLineEncoder(nn.Module):
        def __init__(self, patch_encoder, patches_per_line):
            super().__init__()
            self.patch_encoder = patch_encoder

        def forward(self, x):
            B, K, C, H, W = x.shape
            
            # Flatten batch and patch dimensions to process all patches simultaneously
            x = x.reshape(B * K, C, H, W)
            E = self.patch_encoder(x)
            
            # Reshape back to separate the K patches per line
            E = E.reshape(B, K, -1)
            
            # Mean pooling across K patches
            pooled = E.mean(dim=1)
            
            # Final L2 normalization of the pooled line embedding
            return F.normalize(pooled, p=2, dim=1)
            
    return FastLineEncoder(patch_encoder, patches_per_line)


def build_triplet_siamese(line_encoder: nn.Module, embedding_dim=128, margin=0.3):
    """
    Builds the Siamese network and the Triplet Margin Loss function.
    """
    class TripletSiamese(nn.Module):
        def __init__(self, line_encoder, embedding_dim):
            super().__init__()
            self.line_encoder = line_encoder
            self.embedding_dim = embedding_dim

        def forward(self, a_in, p_in, n_in):
            # Parallel branch execution
            return torch.cat([self.line_encoder(a_in), 
                              self.line_encoder(p_in), 
                              self.line_encoder(n_in)], dim=-1)

    model = TripletSiamese(line_encoder, embedding_dim)

    def triplet_loss(_, y_pred):
        # Slice the concatenated output back into Anchor, Positive, Negative
        a = y_pred[:, :embedding_dim]
        p = y_pred[:, embedding_dim:2*embedding_dim]
        n = y_pred[:, 2*embedding_dim:]
        
        # Fast vector distance calculation
        pos_dist = torch.sum((a - p) ** 2, dim=1)
        neg_dist = torch.sum((a - n) ** 2, dim=1)
        
        # Max(0, pos - neg + margin)
        return torch.clamp(pos_dist - neg_dist + margin, min=0.0).mean()

    return model, triplet_loss
