runtime:
  # NOTE: train.py reads seed from training.seed

preprocessing:
  target_height: 128                    # What it is: Standard height in pixels for all line images after resizing. Aspect ratio is preserved.
                                        # Effect: Increasing (e.g., to 160) may capture finer details but increases computational cost. Decreasing may lose information.
  max_width: 1580                       # What it is: Width of the canvas onto which the resized line is placed.
                                        # Effect: Increasing gives very long lines more space, preventing distortion. Only impactful if your dataset has extremely long lines.
  patch_size: 128                       # What it is: The square dimension (patch_size x patch_size) of the image tiles fed to the model.
                                        # Effect: Must match the model's expected input size. A larger patch sees more context but requires a different model architecture.
  stride: 56                            # What it is: The step size (in pixels) for sliding the patch window across the line image.
                                        # Effect: A smaller stride increases patch overlap, acting as data augmentation. This can improve accuracy but significantly slows down processing.
  min_foreground: 0.04                  # What it is: The minimum fraction of "ink" a patch must have to be kept.  However during prediction of individual lines I found 0.04 is the best
                                        # Effect: Increasing this makes the filter stricter, forcing the model to train on information-rich patches only, which can reduce noise and improve accuracy.
  place_train: "random"                 # What it is: How to place shorter lines on the wide canvas during training.
                                        # Effect: "random" is a crucial data augmentation step. Changing this to "center" or "left" will likely decrease model robustness and accuracy.
  place_eval: "center"                  # What it is: How to place lines during evaluation/testing.
                                        # Effect: "center" provides deterministic and consistent results, which is essential for reliable evaluation. Do not change.
  use_illum: false                      # What it is: Enables illumination correction to handle shadows or uneven lighting.
                                        # Effect: Set to `true` if your scans have shadows. Can significantly improve accuracy on low-quality scans but may be unnecessary for clean data.
  use_clahe: false                      # What it is: Enables local contrast enhancement (CLAHE).
                                        # Effect: Set to `true` if ink is faded or contrast is poor. Can improve accuracy by making strokes more visible.

training:
  seed: 42                              # What it is: The random seed for PyTorch, ensuring training runs are reproducible.
                                        # Effect: Changing this will lead to different weight initializations and data shuffling, resulting in a slightly different final model.
  epochs: 50                            # What it is: The maximum number of times the model will iterate over the entire training dataset.
                                        # Effect: Too few epochs leads to underfitting; too many can lead to overfitting. Often used with early stopping.
  batch_size: 16                        # What it is: The number of triplets processed in a single training step.
                                        # Effect: Larger batches provide more stable gradients but use more memory. Smaller batches can sometimes help generalization.
  learning_rate: 0.0001                 # What it is: The step size for the optimizer. One of the most critical hyperparameters.
                                        # Effect: Too high can cause unstable training; too low can cause very slow or stalled training.
  embedding_dim: 512                    # What it is: The size of the feature vector (embedding) that the model produces for each patch.
                                        # Effect: A larger dimension can capture more information but increases model size and may overfit. A smaller dimension is faster but may be less expressive.
  triplet_margin: 0.3                   # What it is: The required distance between "same writer" pairs and "different writer" pairs in the triplet loss.
                                        # Effect: A larger margin makes the training task harder, forcing more separation. A smaller margin is easier to satisfy. This is a key parameter to tune.
  patches_per_line: 8                   # What it is: The number of patches randomly sampled from each line image to form its overall embedding.
                                        # Effect: More patches provide a more robust representation of the line but slow down training. Fewer patches are faster but may be less representative.
  early_stopping_patience: 10           # What it is: How many epochs to wait for the validation loss to improve before stopping training.
                                        # Effect: A smaller value stops training faster, preventing overfitting. A larger value gives the model more chances to escape a temporary plateau.
  val_split_by_writers: 0.2             # What it is: The fraction of writers from the training set to hold out for validation (e.g., 0.2 = 20%).
                                        # Effect: A larger split gives a more reliable validation metric but reduces the amount of data available for training.
  num_workers: 0                        # What it is: The number of CPU processes to use for loading data in parallel.
                                        # Effect: Increasing this (e.g., to 4) can speed up training if your CPU is a bottleneck, but `0` (main process only) is the most stable setting.

evaluation:
  metric: "cosine"                      # What it is: The distance metric for comparing embeddings. Cosine is excellent for high-dimensional vectors.
                                        # Effect: This is an architectural choice. While cosine is standard, other metrics like Euclidean ('L2') could be used.
  default_threshold: 0.17               # What it is: An initial guess for the decision threshold.
                                        # Effect: Has no impact on the final result, as the evaluation script sweeps a range of thresholds to find the optimal one.
  thresholds:
    start: 0.00                         # The beginning of the range to test for the optimal threshold.
    stop: 2.00                          # The end of the range.
    step: 0.002                         # The increment for testing thresholds. A smaller step is more precise but makes the evaluation process slower.
